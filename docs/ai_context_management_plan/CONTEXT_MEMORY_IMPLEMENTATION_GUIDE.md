currentTokens,\n        isTruncated: false,\n      };\n    }\n    \n    // Step 1: Try compaction\n    let optimized = await this.compact(messages, availableTokens);\n    let optimizedTokens = this.estimateTokens(optimized.map(m => m.content).join('\\n'));\n    \n    if (optimizedTokens <= availableTokens) {\n      return {\n        optimizedMessages: optimized,\n        availableTokens: availableTokens - optimizedTokens,\n        isTruncated: true,\n      };\n    }\n    \n    // Step 2: Try truncation\n    optimized = this.truncate(optimized, availableTokens);\n    optimizedTokens = this.estimateTokens(optimized.map(m => m.content).join('\\n'));\n    \n    return {\n      optimizedMessages: optimized,\n      availableTokens: availableTokens - optimizedTokens,\n      isTruncated: true,\n    };\n  }\n  \n  /**\n   * Compact conversation by summarizing old messages\n   */\n  private async compact(\n    messages: ConversationMessage[],\n    availableTokens: number\n  ): Promise<ConversationMessage[]> {\n    const keepRecent = 5;\n    \n    if (messages.length <= keepRecent) {\n      return messages;\n    }\n    \n    const oldMessages = messages.slice(0, -keepRecent);\n    const recentMessages = messages.slice(-keepRecent);\n    \n    // Summarize old messages\n    const summary = await this.summarizer.summarize(oldMessages, {\n      level: 'brief',\n      maxTokens: availableTokens / 2,\n    });\n    \n    // Create synthetic summary message\n    const summaryMessage: ConversationMessage = {\n      id: `SUMMARY#${Date.now()}`,\n      timestamp: oldMessages[0].timestamp,\n      role: 'system',\n      content: `[CONVERSATION SUMMARY]\\n${summary.text}`,\n      metadata: {\n        tokens: summary.tokenCount,\n      },\n    };\n    \n    return [summaryMessage, ...recentMessages];\n  }\n  \n  /**\n   * Truncate conversation\n   */\n  private truncate(\n    messages: ConversationMessage[],\n    availableTokens: number\n  ): ConversationMessage[] {\n    // Keep recent messages with priority\n    const result: ConversationMessage[] = [];\n    let totalTokens = 0;\n    \n    // Start from the end and work backwards\n    for (let i = messages.length - 1; i >= 0; i--) {\n      const msg = messages[i];\n      const msgTokens = this.estimateTokens(msg.content);\n      \n      if (totalTokens + msgTokens <= availableTokens) {\n        result.unshift(msg);\n        totalTokens += msgTokens;\n      } else {\n        break;\n      }\n    }\n    \n    return result;\n  }\n  \n  /**\n   * Estimate tokens (rough: 1 token ≈ 4 characters)\n   */\n  private estimateTokens(text: string): number {\n    return Math.ceil(text.length / 4);\n  }\n}\n```\n\n---\n\n## 7.
